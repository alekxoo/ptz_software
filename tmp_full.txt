def _now() -> datetime:
    # project asked for UTC minus 5h in earlier iterations; keep UTC and
    # subtract display-side if needed. If you truly want UTC-5 persisted:
    return datetime.now(timezone.utc)  # or: - timedelta(hours=5)

def _parse_rfc3339(s: Optional[str]) -> Optional[datetime]:
    if not s:
        return None
    try:
        if s.endswith("Z"):
            s = s[:-1] + "+00:00"
        dt = datetime.fromisoformat(s)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        return dt
    except Exception:
        return None

def _connected_from_last_seen(last_seen: Optional[datetime], now: Optional[datetime] = None) -> bool:
    if last_seen is None:
        return False
    now = now or _now()
    return (now - last_seen).total_seconds() <= TS_CONNECTED_WINDOW_SECONDS

def _last_value(series: dict) -> Optional[float]:
    """Extract the last scalar value from a series like 'battery_level', etc."""
    try:
        data = series.get("data", [])
        if not data:
            return None

        # Get the last entry: [timestamp, [value]]
        _, val = data[-1]

        # If it's a list (like [0.0]), get the first element
        if isinstance(val, list) and len(val) > 0:
            val = val[0]

        return float(val)
    except Exception as e:
        print(f"_last_value error: {e}")
        return None

    

def _add_unit_once(val: Optional[str], unit: str) -> Optional[str]:
    if val is None:
        return None
    s = str(val).strip().strip('"').strip()
    if s.lower().endswith(unit.lower()):
        s = s[: -len(unit)].strip()
    return f"{s}{unit}"

def _save_device_doc(doc: dict):
    """Persist device document to snapshots and last-known."""
    if col_snap is not None:
        try:
            col_snap.insert_one({**doc})
        except Exception:
            pass
    if col_latest is not None:
        try:
            col_latest.replace_one({"device_id": doc.get("device_id")}, {**doc}, upsert=True)
        except Exception:
            pass

def static_devices() -> List[Dict[str, Any]]:
    try:
        items = json.loads(DEVICES_JSON)
        return items if isinstance(items, list) else []
    except Exception:
        return []

# TAILSCALE DISCOVERY (with lastSeen + connected)
# ──────────────────────────────────────────────────────────────────────────────
async def tailscale_list_devices() -> List[Dict[str, Any]]:
    if not TS_API_KEY or not TS_TAILNET:
        return []

    url = f"{TS_API_BASE}/tailnet/{TS_TAILNET}/devices"
    now = _now()

    async with httpx.AsyncClient(timeout=25.0, auth=(TS_API_KEY, "")) as cx:
        r = await cx.get(url)
        r.raise_for_status()
        data = r.json()

    out: List[Dict[str, Any]] = []
    for dev in data.get("devices", []):
        addrs = dev.get("addresses", []) or []
        ip = next((a for a in addrs if a.startswith("100.")), (addrs[0] if addrs else None))
        if not ip:
            continue

        last_seen_dt = _parse_rfc3339(dev.get("lastSeen"))
        connected = dev.get("online") is True or _connected_from_last_seen(last_seen_dt, now)

        item = {
            "id": dev.get("id") or dev.get("name") or ip,
            "name": dev.get("name") or ip,
            "device_id": (dev.get("name") or ip),
            "tailscaleIp": ip,
            "os": dev.get("os"),
            "lastSeen": last_seen_dt,
            "connected": bool(connected),
        }
        out.append(item)

        if col_devices is not None:
            col_devices.update_one(
                {"tailscaleIp": item["tailscaleIp"]},
                {"$set": {
                    "device_id": item["device_id"],
                    "name": item["name"],
                    "tailscaleIp": item["tailscaleIp"],
                    "os": item["os"],
                    "lastSeen": item["lastSeen"],
                    "connected": item["connected"],
                    "updatedAt": now,
                }},
                upsert=True
            )
    return out


async def discover_devices() -> List[Dict[str, Any]]:
    """Merge static + tailscale (tailscale wins on same IP)"""
    merged: Dict[str, Dict[str, Any]] = {}
    for d in static_devices():
        key = d.get("tailscaleIp") or d.get("id")
        if key:
            merged[key] = d
    for d in await tailscale_list_devices():
        key = d.get("tailscaleIp") or d.get("id")
        if key:
            merged[key] = d
    return list(merged.values())

async def poll_device_combined(device_id: str, device_name: str, ip: str):
    now = _now()

    # ---- IP Webcam sensors
    batt_level_s = batt_temp_s = batt_charging_s = None
    batt_level_num = batt_temp_num = None
    is_charging: Optional[bool] = None

    try:
        sensors_url = f"http://{ip}:{WEBCAM_PORT}/sensors.json?sense=battery_charging,battery_level,battery_temp"
        async with httpx.AsyncClient(timeout=10.0) as cx:
            r = await cx.get(sensors_url)
            r.raise_for_status()
            j = r.json()

        level = _last_value(j.get("battery_level", {}))
        temp  = _last_value(j.get("battery_temp", {}))
        chg   = _last_value(j.get("battery_charging", {}))
        # print(json.dumps(j, indent=2))
        batt_level_num = level
        batt_temp_num  = temp
        is_charging    = True if (chg is not None and float(chg) > 0.0) else (False if chg is not None else None)

        # batt_level_num = f"{int(level)}" if level is not None else None
        # batt_temp_num = f"{round(temp, 1)}" if temp is not None else None
        # is_charging = (
        #     "connected" if is_charging is True else ("disconnected" if is_charging is False else None)
        # )


    except Exception as e:
        if col_alerts is not None:
            col_alerts.insert_one({
                "device_id": device_name,
                "level": "warning",
                "message": f"IP Webcam sensors fetch failed: {e}",
                "timestamp": now
            })

    # ---- Jackery from phone-server (CSV "utc,tempC,inputW,outputW")
    jackery_str = {"temperature": None, "input_power": None, "output_power": None}
    try:
        # NOTE: replace the IP below with your actual phone-server IP or look it up from registry
        url = f"http://100.121.57.115:{PHONE_SVR_PORT}/LastJackeryData"
        async with httpx.AsyncClient(timeout=10.0) as cx:
            r = await cx.get(url)
            r.raise_for_status()
            line = r.text.strip()

        parts = [p.strip().strip('"') for p in line.split(",")]
        temp_c = parts[1] if len(parts) > 1 else None
        in_w   = parts[2] if len(parts) > 2 else None
        out_w  = parts[3] if len(parts) > 3 else None

        jackery_str["temperature"]  = _add_unit_once(temp_c, "℃")
        jackery_str["input_power"]  = _add_unit_once(in_w, "W")
        jackery_str["output_power"] = _add_unit_once(out_w, "W")

        if col_power is not None:
            def _num(s, unit):
                if s is None: return None
                try: return float(str(s).strip().strip('"').replace(unit, '').strip())
                except: return None
            col_power.insert_one({
                "device_id": device_name,
                "ts": now,
                "temp_c": _num(temp_c, "℃"),
                "solar_w": _num(in_w, "W"),
                "load_w": _num(out_w, "W"),
                "raw": {"csv": line},
            })

    except Exception as e:
        if col_alerts is not None:
            col_alerts.insert_one({
                "device_id": device_name,
                "level": "warning",
                "message": f"Jackery LastJackeryData fetch failed: {e}",
                "timestamp": now
            })

    # ---- final device doc (same shape your UI expects)
    device_doc = {
        "device_id": device_name,
        "timestamp": now,
        "battery_level_num": batt_level_num,
        "battery_temp_num": batt_temp_num,
        "is_charging": is_charging,
        "JackeryData": jackery_str,
        "network_type": "tailscale",
    }

    try:
        if col_status is not None:
            col_status.insert_one({**device_doc})
    except Exception as e:
        if col_alerts is not None:
            col_alerts.insert_one({
                "device_id": device_name,
                "level": "warning",
                "message": f"status insert failed: {e}",
                "timestamp": now
            })

    _save_device_doc(device_doc)

# BACKGROUND POLLER
# ──────────────────────────────────────────────────────────────────────────────
async def poll_loop():
    while True:
        try:
            # Refresh the list of devices
            devices = await discover_devices()
            if not devices:
                devices = static_devices()

            tasks = []
            for d in devices:
                device_id = d.get("device_id") or d.get("id")
                ip = d.get("tailscaleIp")
                dev_name = d.get("name") or device_id or ip

                # Require an IP to poll
                if not ip:
                    continue

                # Respect tracking flag, defaulting to ON so devices are polled
                track_on = tracking_state.get(device_id, DEFAULT_TRACKING_ON)
                if not track_on:
                    continue

                # Optional: skip if Tailscale says it's disconnected
                if d.get("connected") is False:
                    continue

                # Track this device
                tasks.append(poll_device_combined(device_id, dev_name, ip))

            # Run all tracking tasks in parallel
            if tasks:
                await asyncio.gather(*tasks)

        except Exception as e:
            if col_alerts is not None:
                col_alerts.insert_one({
                    "device_id": "poller",
                    "level": "warning",
                    "message": f"poll loop error: {e}",
                    "timestamp": _now()
                })

        await asyncio.sleep(POLL_INTERVAL_SECONDS)

@app.get("/api/devices")
def api_devices():
    now = _now()
    items = list(col_latest.find({}, {"_id": 0})) if col_latest is not None else []
    # join with discovery cache for connected/lastSeen
    disc_map: Dict[str, Dict[str, Any]] = {}
    try:
        for d in db.devices.find({}, {"_id": 0, "name": 1, "device_id": 1, "tailscaleIp": 1, "connected": 1, "lastSeen": 1}):
            key = (d.get("device_id") or d.get("name") or d.get("tailscaleIp") or "").lower()
            disc_map[key] = d
    except Exception:
        pass
    
    for d in items:
        ts = d.get("timestamp")
        if isinstance(ts, str):
            try: ts = datetime.fromisoformat(ts)
            except: ts = now
        if ts and ts.tzinfo is None:
            ts = ts.replace(tzinfo=timezone.utc)
        age = (now - ts).total_seconds() if ts else 1e9
        d["status"] = "online" if age <= DEVICE_STALENESS_SECONDS else "offline"

        key = (d.get("device_id") or d.get("name") or "").lower()
        disc = disc_map.get(key)
        if disc:
            d["connected"] = bool(disc.get("connected", False))
            ls = disc.get("lastSeen")
            if isinstance(ls, datetime):
                ls = ls.isoformat()
            d["lastSeen"] = ls

    items.sort(key=lambda x: (x.get("device_id") or x.get("name") or ""))
    return items

@app.get("/api/devices/discovered")
def list_discovered():
    """Return discovery cache with connected + lastSeen and whether registered."""
    existing = {d.get("device_id") for d in db.devices.find({}, {"_id": 0, "device_id": 1})}
    items = list(db.devices.find({}, {"_id": 0}))
    for it in items:
        ls = it.get("lastSeen")
        if isinstance(ls, datetime):
            it["lastSeen"] = ls.isoformat()
        it["registered"] = it.get("device_id") in existing
        it["connected"] = bool(it.get("connected", False))
        it["name"] = it.get("name") or it.get("device_id") or it.get("tailscaleIp")
    items.sort(key=lambda x: (not x.get("connected", False), x.get("name", "")))
    return items

@app.get("/api/devices/{device_id}/logs")
def get_device_logs(device_id: str):
    if col_snap is None:
        raise HTTPException(status_code=500, detail="Log collection not found")

    query = {"device_id": device_id, "battery level": {"$exists": True}, "timestamp": {"$exists": True}}
    projection = {"_id": 0, "battery level": 1, "timestamp": 1}

    try:
        cursor = col_snap.find(query, projection).sort("timestamp", DESCENDING).limit(100)
        logs = list(cursor)[::-1]  # return oldest → newest

        for log in logs:
            # Convert string to int
            battery_str = log.get("battery level")
            log["battery_level_num"] = int(battery_str) if battery_str is not None else None

        return json.loads(dumps(logs))
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Failed to fetch logs: {e}")

# field -> (numeric field in col_status, default aggregation)
FIELD_MAP: Dict[str, str] = {
    "battery_percentage": "battery_level_num",
    "battery_temperature": "battery_temp_num",
    "latency_manual_ms": "latency_manual_ms",
}

def _parse_range(s: str) -> timedelta:
    # e.g. "6h", "30m", "1d"
    unit = s[-1].lower()
    n = int(s[:-1])
    return timedelta(hours=n) if unit == "h" else timedelta(minutes=n) if unit == "m" else timedelta(days=n)

def _parse_interval_to_seconds(s: str) -> int:
    unit = s[-1].lower()
    n = int(s[:-1])
    return n * 3600 if unit == "h" else n * 60 if unit == "m" else n

@app.get("/api/metrics")
def api_metrics(
    device_id: str = Query(...),
    field: str = Query(...),
    range: str = Query("1h"),
    interval: str = Query("1m"),
):
    num_field = FIELD_MAP.get(field)
    if not num_field:
        return {"points": []}

    now = _now()
    start = now - _parse_range(range)
    bucket_sec = _parse_interval_to_seconds(interval)

    pipeline = [
        {"$match": {
            "device_id": device_id,
            "timestamp": {"$gte": start, "$lte": now},
            num_field: {"$ne": None}
        }},
        {"$addFields": {
            "bucket": {
                "$toDate": {
                    "$multiply": [
                        {"$floor": {"$divide": [{"$toLong": "$timestamp"}, bucket_sec * 1000]}},
                        bucket_sec * 1000
                    ]
                }
            }
        }},
        {"$group": {"_id": "$bucket", "avg": {"$avg": f"${num_field}"}}},
        {"$project": {"t": "$_id", "avg": 1, "_id": 0}},
        {"$sort": {"t": 1}},
    ]
    rows = list(col_status.aggregate(pipeline)) if col_status else []
    # ensure ISO times for frontend
    for r in rows:
        if isinstance(r.get("t"), datetime):
            r["t"] = r["t"].isoformat()
    return {"points": rows}
